cmake_minimum_required(VERSION 3.24)
project(zfm_elf LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=4)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK compile_pool)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LIEF_ELF ON CACHE BOOL "" FORCE)
foreach(option PE COFF MACHO DEX ART OAT VDEX C_API PYTHON_API EXAMPLES TESTS LOGGING LOGGING_DEBUG ENABLE_JSON USE_CCACHE INSTALL)
    set(LIEF_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LIEF_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${ZFM_LIEF_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "LIEF source missing. Run the Gradle prepareLief task.")
endif()
add_subdirectory("${ZFM_LIEF_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/lief" EXCLUDE_FROM_ALL)
target_compile_options(LIB_LIEF PRIVATE -O2 -g0)
add_library(zfm_elf SHARED elf_patch.cpp elf_jni.cpp)
target_link_libraries(zfm_elf PRIVATE LIEF::LIEF log)
target_compile_options(zfm_elf PRIVATE -fvisibility=hidden -Wall -Wextra)
target_link_options(zfm_elf PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)

# Used only by the native regression harness, never packaged in the manager.
if(ZFM_PATCH_TESTS)
    add_executable(elf_patch_test elf_patch.cpp elf_patch_test.cpp)
    target_link_libraries(elf_patch_test PRIVATE LIEF::LIEF dl log)
    target_link_options(elf_patch_test PRIVATE -Wl,-z,max-page-size=16384)
    add_library(patch_fixture SHARED patch_fixture.cpp)
    target_link_options(patch_fixture PRIVATE -Wl,-z,max-page-size=16384)
    add_library(patch_dependency SHARED patch_dependency.cpp)
    target_link_options(patch_dependency PRIVATE -Wl,-z,max-page-size=16384)
endif()
